rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Allow users to read/write their own userStats
    match /userStats/{userId} {
      allow read, update: if request.auth != null && request.auth.uid == userId;
      allow create: if request.auth != null && request.auth.uid == userId;
    }
    
    // Schools Collection
    match /schools/{schoolId} {
      // Allow reading school info if user is part of that school
      allow read: if request.auth != null && 
                  (resource.data.teachers.hasAny([request.auth.uid]) || 
                   resource.data.students.hasAny([request.auth.uid]) ||
                   resource.data.admins.hasAny([request.auth.uid]));
      
      // Only admins can create/update schools
      allow write: if request.auth != null && 
                   resource.data.admins.hasAny([request.auth.uid]);
    }

    // Teachers Collection
    match /teachers/{teacherId} {
      // Teachers can read their own profile
      // School admins can read any teacher in their school
      allow read: if request.auth != null && 
                  (request.auth.uid == teacherId || 
                   get(/databases/$(database)/documents/schools/$(resource.data.schoolId)).data.admins.hasAny([request.auth.uid]));
      
      // Teachers can update their own profile
      // School admins can update any teacher in their school
      allow write: if request.auth != null && 
                   (request.auth.uid == teacherId || 
                    get(/databases/$(database)/documents/schools/$(resource.data.schoolId)).data.admins.hasAny([request.auth.uid]));
    }

    // Students Collection
    match /students/{studentId} {
      // Students can read their own profile
      // Teachers can read their students' profiles
      // School admins can read any student in their school
      allow read: if request.auth != null && 
                  (request.auth.uid == studentId || 
                   get(/databases/$(database)/documents/teachers/$(request.auth.uid)).data.schoolId == resource.data.schoolId ||
                   get(/databases/$(database)/documents/schools/$(resource.data.schoolId)).data.admins.hasAny([request.auth.uid]));
      
      // Students can update their own profile
      // School admins can update any student in their school
      allow write: if request.auth != null && 
                   (request.auth.uid == studentId || 
                    get(/databases/$(database)/documents/schools/$(resource.data.schoolId)).data.admins.hasAny([request.auth.uid]));
    }

    // Classes Collection
    match /classes/{classId} {
      // Teachers and students can view their class info
      // School admins can view any class in their school
      allow read: if request.auth != null && 
                  (resource.data.teacherId == request.auth.uid || 
                   resource.data.students.hasAny([request.auth.uid]) ||
                   get(/databases/$(database)/documents/schools/$(resource.data.schoolId)).data.admins.hasAny([request.auth.uid]));
      
      // Only school admins and assigned teachers can update class info
      allow write: if request.auth != null && 
                   (resource.data.teacherId == request.auth.uid || 
                    get(/databases/$(database)/documents/schools/$(resource.data.schoolId)).data.admins.hasAny([request.auth.uid]));
    }

    // User roles - only readable by the user themselves or school admins
    match /user_roles/{userId} {
      allow read: if request.auth != null && 
                  (request.auth.uid == userId || 
                   get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin');
    }

    // Cognitive training sessions
    match /cognitiveSessions/{sessionId} {
      allow create: if request.auth != null && 
                    request.resource.data.userId == request.auth.uid;
      allow read: if request.auth != null && 
                   resource.data.userId == request.auth.uid;
      allow update: if request.auth != null && 
                     resource.data.userId == request.auth.uid &&
                     request.resource.data.diff(resource.data).affectedKeys().hasOnly(['endTime']);
    }
  
    // Game results
    match /game_results/{docId} {
      allow read, update, delete: if request.auth != null && 
                                  request.auth.uid == resource.data.user_id;
      allow create: if request.auth != null && 
                     request.auth.uid == request.resource.data.user_id;
    }

    // Feedback collection
    match /platform_feedback/{docId} {
      allow read, update, delete: if request.auth != null && 
                                  request.auth.uid == resource.data.user_id;
      allow create: if request.auth != null && 
                     request.auth.uid == request.resource.data.user_id;
    }

    // Journal entries (WELLNESS JOURNAL)
    match /journal_entries/{docId} {
      allow read, update, delete: if request.auth != null && 
                                  request.auth.uid == resource.data.user_id;
      allow create: if request.auth != null && 
                     request.auth.uid == request.resource.data.user_id;
    }

    // Assessment responses
    match /assessment_responses/{docId} {
      allow read, update, delete: if request.auth != null && 
                                  request.auth.uid == resource.data.user_id;
      allow create: if request.auth != null && 
                     request.auth.uid == request.resource.data.user_id;
    }

    // Mindfulness sessions
    match /mindfulness_sessions/{docId} {
      allow read, update, delete: if request.auth != null && 
                                  request.auth.uid == resource.data.user_id;
      allow create: if request.auth != null && 
                     request.auth.uid == request.resource.data.user_id;
    }

    // Check-ins
    match /check_ins/{docId} {
      allow read, update, delete: if request.auth != null && 
                                  request.auth.uid == resource.data.user_id;
      allow create: if request.auth != null && 
                     request.auth.uid == request.resource.data.user_id;
    }

    // Demographics (one per user)
    match /demographics/{userId} {
      allow read: if request.auth != null && 
                  request.auth.uid == userId;
      allow create: if request.auth != null && 
                    request.auth.uid == userId && 
                    !exists(/databases/$(database)/documents/demographics/$(userId));
      allow update: if request.auth != null && 
                    request.auth.uid == userId;
    }

    // Mood entries
    match /mood_entries/{docId} {
      allow read, update, delete: if request.auth != null && 
                                  request.auth.uid == resource.data.user_id;
      allow create: if request.auth != null && 
                     request.auth.uid == request.resource.data.user_id;
    }

    // Wellness goals
    match /wellness_goals/{docId} {
      allow read, update, delete: if request.auth != null && 
                                  request.auth.uid == resource.data.user_id;
      allow create: if request.auth != null && 
                     request.auth.uid == request.resource.data.user_id;
    }

    // Reminders
    match /reminders/{docId} {
      allow read, update, delete: if request.auth != null && 
                                  request.auth.uid == resource.data.user_id;
      allow create: if request.auth != null && 
                     request.auth.uid == request.resource.data.user_id;
    }

    // Achievements
    match /user_achievements/{docId} {
      allow read, update, delete: if request.auth != null && 
                                  request.auth.uid == resource.data.user_id;
      allow create: if request.auth != null && 
                     request.auth.uid == request.resource.data.user_id;
    }

    // Anonymous forum posts (read allowed to all, write only to owner)
    match /forum_posts/{docId} {
      allow read: if true;
      allow update, delete: if request.auth != null && 
                            request.auth.uid == resource.data.user_id;
      allow create: if request.auth != null && 
                     request.auth.uid == request.resource.data.user_id;
    }

    // Forum replies (read allowed to all, write only to owner)
    match /forum_replies/{docId} {
      allow read: if true;
      allow update, delete: if request.auth != null && 
                            request.auth.uid == resource.data.user_id;
      allow create: if request.auth != null && 
                     request.auth.uid == request.resource.data.user_id;
    }

    // BuddySafe incident reports (allow anyone to create, read only for admin)
    match /incidents/{docId} {
      allow create: if true;
      allow read, update, delete: if false;
    }

    // User-specific collections
    match /users/{userId} {
      allow read, write: if request.auth != null && 
                         request.auth.uid == userId;
    }

    // Default: deny all other operations
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
